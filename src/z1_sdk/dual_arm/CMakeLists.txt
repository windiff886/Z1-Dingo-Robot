cmake_minimum_required(VERSION 3.16)
project(z1_official_dual_arm VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(Z1_BUILD_PYTHON "Build the official Python bindings with dual-arm ports" OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|aarch64)$")
  message(FATAL_ERROR "The official Z1 libraries require Linux x86_64 or aarch64")
endif()
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(Boost REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include(GNUInstallDirs)
get_filename_component(SDK "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(CONTROLLER "${SDK}/controller")
set(SDK_LIBRARY "${SDK}/lib/libZ1_SDK_${CMAKE_SYSTEM_PROCESSOR}.so")
set(CONTROLLER_LIBRARY "${CONTROLLER}/lib/libZ1_${CMAKE_SYSTEM_PROCESSOR}.so")
add_library(z1_official_sdk SHARED IMPORTED GLOBAL)
set_target_properties(z1_official_sdk PROPERTIES IMPORTED_LOCATION "${SDK_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${SDK}/include"
  INTERFACE_LINK_LIBRARIES "Eigen3::Eigen;Threads::Threads")
add_library(z1_official::sdk ALIAS z1_official_sdk)
add_library(z1_official_controller SHARED IMPORTED)
set_target_properties(z1_official_controller PROPERTIES IMPORTED_LOCATION "${CONTROLLER_LIBRARY}")

# Keep upstream files verbatim. Generate only a startup port adaptation.
file(READ "${CONTROLLER}/main.cpp" CONTROLLER_MAIN)
set(PORTS_OLD "\"127.0.0.1\", 8072, 8071, 0.002")
set(PORTS_NEW "\"127.0.0.1\", z1_environment_port(\"Z1_SDK_PORT\", 8072), z1_environment_port(\"Z1_CONTROLLER_PORT\", 8071), 0.002")
string(FIND "${CONTROLLER_MAIN}" "${PORTS_OLD}" PORTS_POSITION)
if(PORTS_POSITION LESS 0)
  message(FATAL_ERROR "Upstream controller startup changed; review the port adaptation")
endif()
string(REPLACE "${PORTS_OLD}" "${PORTS_NEW}" CONTROLLER_MAIN "${CONTROLLER_MAIN}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/official_controller_main.cpp"
  CONTENT "#include \"controller_environment.hpp\"\n${CONTROLLER_MAIN}")
add_executable(z1_ctrl "${CMAKE_CURRENT_BINARY_DIR}/official_controller_main.cpp")
set_target_properties(z1_ctrl PROPERTIES CXX_STANDARD 14)
target_include_directories(z1_ctrl PRIVATE src "${CONTROLLER}/include" ${Boost_INCLUDE_DIRS})
target_link_libraries(z1_ctrl PRIVATE z1_official_controller Eigen3::Eigen Threads::Threads)
add_executable(z1_sdk_client src/sdk_client.cpp)
target_include_directories(z1_sdk_client PRIVATE include)
target_link_libraries(z1_sdk_client PRIVATE z1_official::sdk)
target_compile_options(z1_sdk_client PRIVATE -Wall -Wextra)

if(Z1_BUILD_PYTHON)
  find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
  set(PYTHON_EXECUTABLE "${Python3_EXECUTABLE}" CACHE FILEPATH "Python for pybind11")
  find_package(pybind11 REQUIRED CONFIG)
  file(READ "${SDK}/examples_py/arm_python_interface.cpp" PYTHON_BINDING)
  set(CONSTRUCTOR_OLD "ArmInterface(bool hasGripper):unitreeArm(hasGripper){};")
  set(CONSTRUCTOR_NEW "ArmInterface(bool hasGripper, unsigned controllerPort, unsigned sdkPort, double load):unitreeArm(z1_official::components(controllerPort, sdkPort, hasGripper, load)){};")
  string(FIND "${PYTHON_BINDING}" "${CONSTRUCTOR_OLD}" CONSTRUCTOR_POSITION)
  if(CONSTRUCTOR_POSITION LESS 0)
    message(FATAL_ERROR "Upstream Python constructor changed; review the dual-arm adaptation")
  endif()
  string(REPLACE "${CONSTRUCTOR_OLD}" "${CONSTRUCTOR_NEW}" PYTHON_BINDING "${PYTHON_BINDING}")
  string(REPLACE ".def(py::init<bool>(), py::arg(\"hasGripper\")=true)"
    ".def(py::init<bool, unsigned, unsigned, double>(), py::arg(\"hasGripper\")=true, py::arg(\"controllerPort\")=8071, py::arg(\"sdkPort\")=8072, py::arg(\"load\")=0.0)"
    PYTHON_BINDING "${PYTHON_BINDING}")
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/official_python_interface.cpp"
    CONTENT "#include \"z1_official/sdk.hpp\"\n${PYTHON_BINDING}")
  pybind11_add_module(unitree_arm_interface "${CMAKE_CURRENT_BINARY_DIR}/official_python_interface.cpp")
  target_include_directories(unitree_arm_interface PRIVATE include)
  target_link_libraries(unitree_arm_interface PRIVATE z1_official::sdk)
  set_target_properties(unitree_arm_interface PROPERTIES INSTALL_RPATH "$ORIGIN")
  install(TARGETS unitree_arm_interface LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()

set(BIN_DEST "${CMAKE_INSTALL_BINDIR}")
set(BINARY_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
set(Z1_SOURCE_ROOT "${SDK}")
set(Z1_BUILD_DIR "${CMAKE_BINARY_DIR}/bin")
configure_file(scripts/z1_official.py "${CMAKE_BINARY_DIR}/bin/z1_official.py" @ONLY)
install(PROGRAMS "${CMAKE_BINARY_DIR}/bin/z1_official.py" DESTINATION ${BIN_DEST})
install(DIRECTORY config/ DESTINATION share/z1_sdk/config)
set_target_properties(z1_ctrl z1_sdk_client PROPERTIES INSTALL_RPATH "${BINARY_RPATH}")
install(TARGETS z1_ctrl z1_sdk_client RUNTIME DESTINATION ${BIN_DEST})
set_target_properties(z1_ctrl z1_sdk_client PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(FILES "${SDK_LIBRARY}" "${CONTROLLER_LIBRARY}" DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ "${SDK}/include/" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY "${CONTROLLER}/config/" DESTINATION share/z1_sdk/upstream_config)
install(FILES "${SDK}/LICENSE" DESTINATION share/z1_sdk/licenses RENAME z1_sdk.LICENSE)
install(FILES "${CONTROLLER}/LICENSE" DESTINATION share/z1_sdk/licenses RENAME z1_controller.LICENSE)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/Z1OfficialSDKConfig.cmake.in Z1OfficialSDKConfig.cmake
  INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/Z1OfficialSDK"
  PATH_VARS CMAKE_INSTALL_LIBDIR CMAKE_INSTALL_INCLUDEDIR)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/Z1OfficialSDKConfig.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Z1OfficialSDK)
include(CTest)
