cmake_minimum_required(VERSION 3.16)
project(z1_sdk LANGUAGES CXX)
include(CTest)
add_subdirectory(dual_arm)
option(Z1_BUILD_EXAMPLES "Build all upstream C++ examples (these move hardware when run)" ON)
if(Z1_BUILD_EXAMPLES)
  foreach(name highcmd_basic highcmd_development lowcmd_development lowcmd_multirobots)
    add_executable(${name} examples/${name}.cpp)
    set_target_properties(${name} PROPERTIES CXX_STANDARD 14 RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/examples")
    target_link_libraries(${name} PRIVATE z1_official::sdk)
  endforeach()
endif()
