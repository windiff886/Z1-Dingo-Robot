cmake_minimum_required(VERSION 3.0.2)
project(unidingo_rviz_test)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(catkin REQUIRED COMPONENTS roscpp z1_control geometry_msgs
  std_msgs trajectory_msgs message_generation rviz pluginlib moveit_msgs moveit_ros_visualization)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_service_files(FILES PreviewPose.srv)
add_message_files(FILES PreviewStatus.msg EditorTarget.msg EditorCommand.msg)
generate_messages(DEPENDENCIES std_msgs geometry_msgs trajectory_msgs)
catkin_package(CATKIN_DEPENDS message_runtime std_msgs geometry_msgs trajectory_msgs)
add_executable(z1_pose_preview src/pose_preview.cpp)
target_include_directories(z1_pose_preview PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(z1_pose_preview PRIVATE z1_core ${roscpp_LIBRARIES})
add_dependencies(z1_pose_preview ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
set_target_properties(z1_pose_preview PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
install(TARGETS z1_pose_preview RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
add_library(z1_rviz_plugins SHARED src/target_panel.cpp src/sdk_motion_planning.cpp)
set_target_properties(z1_rviz_plugins PROPERTIES AUTOMOC ON)
target_include_directories(z1_rviz_plugins PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(z1_rviz_plugins PRIVATE ${moveit_ros_visualization_LIBRARIES} ${rviz_LIBRARIES} ${roscpp_LIBRARIES} Qt5::Widgets)
add_dependencies(z1_rviz_plugins ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
install(TARGETS z1_rviz_plugins LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES plugin_description.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
catkin_install_python(PROGRAMS scripts/interactive_tcp_test.py
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config rviz DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
