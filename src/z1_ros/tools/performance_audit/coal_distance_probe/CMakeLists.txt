cmake_minimum_required(VERSION 3.16)
project(coal_distance_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(urdfdom QUIET)
if(TARGET urdfdom::urdfdom_model AND NOT TARGET urdfdom::urdf_parser)
  add_library(urdfdom::urdf_parser ALIAS urdfdom::urdfdom_model)
endif()
find_package(pinocchio 4.0 REQUIRED)
add_executable(probe probe.cpp)
target_link_libraries(probe PRIVATE pinocchio::pinocchio)
