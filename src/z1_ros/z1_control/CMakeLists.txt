cmake_minimum_required(VERSION 3.16)
project(z1_control VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(Z1_BUILD_ROS "Build the ROS 1 topic adapter" OFF)
option(Z1_BUILD_CORE "Build the model and motion control core" ON)
if(DEFINED CATKIN_DEVEL_PREFIX)
  set(Z1_BUILD_ROS ON)
endif()
if(Z1_BUILD_ROS AND NOT Z1_BUILD_CORE)
  message(FATAL_ERROR "Z1_BUILD_ROS requires Z1_BUILD_CORE")
endif()
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(yaml-cpp REQUIRED)
if(NOT TARGET yaml-cpp::yaml-cpp)
  add_library(yaml-cpp::yaml-cpp ALIAS yaml-cpp)
endif()
find_package(Threads REQUIRED)
set(PYTHON_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "Python for Pinocchio discovery")
set(Python3_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "Python for dependencies")
if(Z1_BUILD_CORE)
  find_package(urdfdom QUIET)
  if(TARGET urdfdom::urdfdom_model AND NOT TARGET urdfdom::urdf_parser)
    add_library(urdfdom::urdf_parser ALIAS urdfdom::urdfdom_model)
  endif()
  find_package(pinocchio 4.0 REQUIRED)
  find_package(ruckig 0.9.2 REQUIRED)
  find_package(OpenSSL REQUIRED)
  add_library(z1_core STATIC src/model.cpp src/dynamics.cpp src/collision.cpp src/reference.cpp src/trajectory.cpp src/cartesian_path.cpp src/controller.cpp src/runtime.cpp)
  target_include_directories(z1_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_link_libraries(z1_core PUBLIC pinocchio::pinocchio ruckig::ruckig Threads::Threads OpenSSL::Crypto)
  target_compile_options(z1_core PRIVATE -Wall -Wextra)
  set_target_properties(z1_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
endif()
get_filename_component(VENDOR_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/unitree" ABSOLUTE)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|aarch64)$")
  message(FATAL_ERROR "MCU adapter requires Linux x86_64 or aarch64")
endif()
set(VENDOR_LIB "${VENDOR_DIR}/lib/libUnitreeArm_${CMAKE_SYSTEM_PROCESSOR}.so")
add_library(z1_unitree_vendor SHARED IMPORTED)
set_target_properties(z1_unitree_vendor PROPERTIES IMPORTED_LOCATION "${VENDOR_LIB}")
add_library(z1_mcu_transport SHARED src/mcu_transport.cpp src/observed_udp.cpp)
target_include_directories(z1_mcu_transport PUBLIC include PRIVATE "${VENDOR_DIR}/include" "${VENDOR_DIR}/include/IOInterface")
target_link_libraries(z1_mcu_transport PUBLIC Eigen3::Eigen PRIVATE z1_unitree_vendor)
set_target_properties(z1_mcu_transport PROPERTIES INSTALL_RPATH "$ORIGIN")
add_executable(z1_calibrate src/calibrate.cpp)
target_link_libraries(z1_calibrate PRIVATE z1_mcu_transport yaml-cpp::yaml-cpp Threads::Threads)
target_compile_options(z1_calibrate PRIVATE -Wall -Wextra)
target_compile_definitions(z1_calibrate PRIVATE
  Z1_CALIBRATION_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/calibration.yaml")
set_target_properties(z1_calibrate PROPERTIES
  INSTALL_RPATH "$ORIGIN/../lib" INSTALL_RPATH_USE_LINK_PATH TRUE)
include(CTest)
if(BUILD_TESTING)
  if(Z1_BUILD_CORE)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE z1_core)
    add_test(NAME arm_core COMMAND core_tests "${CMAKE_CURRENT_SOURCE_DIR}/../z1_description/urdf/z1_control.urdf")
    add_executable(reference_tests tests/reference_tests.cpp)
    target_link_libraries(reference_tests PRIVATE z1_core)
    add_test(NAME reference_geometry COMMAND reference_tests "${CMAKE_CURRENT_SOURCE_DIR}/../z1_description/urdf/z1_control.urdf")
    add_executable(runtime_tests tests/runtime_tests.cpp)
    target_link_libraries(runtime_tests PRIVATE z1_core)
    add_test(NAME runtime_timing COMMAND runtime_tests "${CMAKE_CURRENT_SOURCE_DIR}/../z1_description/urdf/z1_control.urdf")
    add_executable(trajectory_tests tests/trajectory_tests.cpp)
    target_link_libraries(trajectory_tests PRIVATE z1_core)
    add_test(NAME timed_trajectory COMMAND trajectory_tests)
    add_executable(cartesian_path_tests tests/cartesian_path_tests.cpp)
    target_link_libraries(cartesian_path_tests PRIVATE z1_core)
    add_test(NAME cartesian_path COMMAND cartesian_path_tests "${CMAKE_CURRENT_SOURCE_DIR}/../z1_description/urdf/z1_control.urdf")
    add_executable(prepared_motion_tests tests/prepared_motion_tests.cpp)
    target_link_libraries(prepared_motion_tests PRIVATE z1_core)
    add_test(NAME prepared_motion COMMAND prepared_motion_tests "${CMAKE_CURRENT_SOURCE_DIR}/../z1_description/urdf/z1_control.urdf")
    add_executable(motion_contract_tests tests/motion_contract_tests.cpp)
    target_link_libraries(motion_contract_tests PRIVATE z1_core)
    add_test(NAME motion_contract COMMAND motion_contract_tests "${CMAKE_CURRENT_SOURCE_DIR}/../z1_description/urdf/z1_control.urdf")
    set_tests_properties(motion_contract PROPERTIES TIMEOUT 30)
    set_tests_properties(arm_core reference_geometry runtime_timing timed_trajectory cartesian_path prepared_motion PROPERTIES TIMEOUT 90)
  endif()
  add_executable(transport_tests tests/transport_tests.cpp)
  target_link_libraries(transport_tests PRIVATE z1_mcu_transport Threads::Threads)
  add_test(NAME mcu_udp_loopback COMMAND transport_tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME dual_arm_calibration COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/calibration_test.py" $<TARGET_FILE:z1_calibrate>)
  set_tests_properties(mcu_udp_loopback dual_arm_calibration PROPERTIES
    TIMEOUT 90 RESOURCE_LOCK z1_mcu_loopback)
endif()
if(Z1_BUILD_ROS)
  find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs geometry_msgs z1_msgs)
  catkin_package(INCLUDE_DIRS include LIBRARIES z1_core z1_mcu_transport CATKIN_DEPENDS roscpp sensor_msgs geometry_msgs z1_msgs)
  add_executable(z1_topic_node src/ros_node.cpp)
  target_include_directories(z1_topic_node PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(z1_topic_node PRIVATE z1_core z1_mcu_transport ${catkin_LIBRARIES})
  add_dependencies(z1_topic_node ${catkin_EXPORTED_TARGETS})
  set_target_properties(z1_topic_node PROPERTIES INSTALL_RPATH "$ORIGIN/.." INSTALL_RPATH_USE_LINK_PATH TRUE)
  install(TARGETS z1_topic_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  set_target_properties(z1_calibrate PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_BIN_DESTINATION}"
    INSTALL_RPATH "$ORIGIN/..")
  install(TARGETS z1_calibrate RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
else()
  install(TARGETS z1_calibrate RUNTIME DESTINATION bin)
  install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../z1_bringup/config/communication.yaml"
    DESTINATION share/z1_bringup/config)
endif()
if(Z1_BUILD_CORE)
  install(TARGETS z1_core ARCHIVE DESTINATION lib)
endif()
install(TARGETS z1_mcu_transport LIBRARY DESTINATION lib)
install(DIRECTORY config DESTINATION share/z1_control)
install(DIRECTORY include/ DESTINATION include)
install(FILES "${VENDOR_LIB}" DESTINATION lib)
install(FILES "${VENDOR_DIR}/LICENSE" DESTINATION share/z1_control/vendor)
